📋 PARAMETER REFERENCE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━


REQUIRED (at least for first run):
  python3 rotate_model_to_gif.py

OPTIONAL PARAMETERS:
  --texture [PATH]    Texture file (PNG/JPG)
  --axis [AXIS]       Rotation axis (default: z)
  --frames [N]        Number of frames (default: 36)
  --fps [N]           Playback speed (default: 10)
  --obj [PATH]        OBJ file (default: 3DModel_1/OBJ/3DModel.obj)
  --output [PATH]     Output file (default: 3DModel_1/model_rotation.gif)
  --help              Show help text


AXIS OPTIONS:
  Standard:   x, y, z
  Arbitrary:  1,1,1 or 0.7,0.5,0.5 (comma-separated values)
  Automatic:  Vectors normalized to unit length

Error: "Module not found"
  Solution: Make sure you've activated the virtual environment
  Command: source venv/bin/activate

Error: "File not found"
  Solution: Check that the texture file exists
  Command: ls 3DModel_1/OBJ/3DModel/*.png

Error: "Invalid axis"
  Solution: Use format: a,b,c (commas, no spaces)
  Example: --axis 1,1,1 (correct)
           --axis 1, 1, 1 (wrong - spaces)

Error: "Output file too large"
  Solution: Reduce frames or fps
  Command: python3 rotate_model_to_gif.py --frames 20 --fps 10

Help:
  Command: python3 rotate_model_to_gif.py --help


✅ WHAT WAS CHANGED (v1.0 → v2.0)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━


CHANGE 1: Texture Parameter
  Before: --texture-dir 3DModel_1/OBJ/3DModel (auto-scanned)
  After:  --texture 3DModel_1/OBJ/3DModel/specific_file.png (direct)

CHANGE 2: Rotation Axis
  Before: --axis x  or  --axis y  or  --axis z  (only 3 options)
  After:  --axis x  or  --axis y  or  --axis z  or  --axis 1,1,1  (unlimited)

CHANGE 3: Background
  Before: 3 visible pane edges visible
  After:  Completely white (no visible edges)
